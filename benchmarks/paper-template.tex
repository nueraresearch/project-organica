\documentclass[11pt,letterpaper]{article}

\usepackage[margin=1in]{geometry}
\usepackage[T1]{fontenc}
% Latin Modern ships scalable Type1 fonts: microtype's expansion needs them,
% and the default T1 set does not have them, which is a fatal error.
\usepackage{lmodern}
% Emit a real Unicode map for the Type1 fonts, so the deposited PDF's text
% extracts as the words on the page (fi/fl, the section sign).
\input{glyphtounicode}
\pdfgentounicode=1
\usepackage[utf8]{inputenc}

% The manuscripts are written with typographic punctuation.  pdfLaTeX has no
% declaration for these characters by default, so each is declared here; the
% mathematical ones are wrapped in \ensuremath so they are correct in text
% and in math mode alike.
\DeclareUnicodeCharacter{00A7}{\S}
\DeclareUnicodeCharacter{00B1}{\ensuremath{\pm}}
\DeclareUnicodeCharacter{00D7}{\ensuremath{\times}}
\DeclareUnicodeCharacter{00F2}{\^o}
\DeclareUnicodeCharacter{03B1}{\ensuremath{\alpha}}
\DeclareUnicodeCharacter{2013}{--}
\DeclareUnicodeCharacter{2014}{---}
\DeclareUnicodeCharacter{2192}{\ensuremath{\rightarrow}}
\DeclareUnicodeCharacter{2212}{\ensuremath{-}}
\DeclareUnicodeCharacter{2248}{\ensuremath{\approx}}
\DeclareUnicodeCharacter{2265}{\ensuremath{\geq}}
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{graphicx}
\usepackage{booktabs}
\usepackage{longtable}
% Six of Paper 02's audit tables are too wide for a portrait measure at any
% readable size: the nine-column ancestry table gives each column 41 pt and
% the header word "Reachability" is 58 pt.  Those tables are typeset on a
% landscape page instead, where the same fraction has room to breathe.
\usepackage{pdflscape}
\usepackage{array}
\usepackage{calc}   % pandoc writes column widths as \real{...}

% Code blocks.  Pandoc wraps a code block that carries a class in a "Shaded"
% environment and gives the block itself a "Highlighting" environment; both
% come from pandoc's own default template, which this one replaces, so they
% are supplied here.  fancyvrb must be loaded first: the highlighting macros
% define "Highlighting" in terms of it.
\usepackage{fancyvrb}
$highlighting-macros$
% Code blocks one size down.  Paper 02's longest comment line is 103 characters
% and ran 61 pt into the margin; breaking it would put a line break into a block
% a reader may copy, so the block is set smaller instead of being rewritten.
\fvset{fontsize=\footnotesize}
\usepackage[font=small,labelfont=bf]{caption}
\usepackage[hidelinks]{hyperref}
% The PDF's own document properties: what a viewer's title bar, a reference
% manager's import and a preprint server's metadata block read.  The values
% arrive as plain text because hyperref's pdfusetitle would copy the typeset
% \title, which is a \parbox here, markup and all.
$if(pdf-title)$
\hypersetup{pdftitle={$pdf-title$},pdfauthor={$pdf-author$},%
  pdfsubject={$pdf-subject$},pdfkeywords={$pdf-keywords$}}
$endif$
% fi and fl ligatures extract as nothing from a T1-encoded PDF, so the deposited text
% reads "con rmed".  Disabling them keeps the text searchable at no visual cost.
\usepackage{microtype}
\DisableLigatures{encoding = T1, family = *}

\setlength{\parskip}{0.35em}
\setlength{\emergencystretch}{3em}
\providecommand{\tightlist}{\setlength{\itemsep}{0pt}\setlength{\parskip}{0pt}}

$if(title)$
\title{\parbox{0.86\textwidth}{\centering $title$}}
$endif$
$if(author)$
\author{\parbox{0.86\textwidth}{\centering $author$}}
$endif$
\date{}

% The manuscripts carry their own numbers in the heading text ("4.10 The confirmed
% capacity test"), so LaTeX must not add a second, conflicting set.
\setcounter{secnumdepth}{0}
\begin{document}
\maketitle
$body$
\end{document}
